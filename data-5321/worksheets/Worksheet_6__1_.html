# Function to run a sentiment classification model for a given vocabulary size
def imdb_data(vocab_size, batch_size=512, epochs=10, maxlen=500):
    (x_train, y_train), (x_test, y_test) = imdb.load_data(num_words=vocab_size)
    x_train = sequence.pad_sequences(x_train, maxlen=maxlen)
    x_test = sequence.pad_sequences(x_test, maxlen=maxlen)

    model = Sequential([
        Embedding(vocab_size, 32, input_length=maxlen),
        GlobalAveragePooling1D(),
        Dense(1, activation='sigmoid')
    ])
    model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])

    history = model.fit(
        x_train, y_train,
        epochs=epochs,
        batch_size=batch_size,
        validation_split=0.2,
        verbose=1
    )

    test_loss, test_accuracy = model.evaluate(x_test, y_test, verbose=0)
    return history, test_loss, test_accuracy

# Function to create lagged features for time series modeling
def create_lags(data, lags=5):
    for i in range(1, lags + 1):
        # For each lag, shift all predictor columns by i days
        data[[f'{col}_lag{i}' for col in cols]] = data[cols].shift(i)
    return data.dropna()  # Remove rows with NaNs caused by shifting

RNN with 7 Lags¶

We extended the RNN model by using 7 lagged time steps instead of 5. This increased the model's input dimensionality and temporal context.

  • Training loss decreased well, indicating learning.
  • Validation (test) R² dropped from 0.9604 to 0.8684.
  • This suggests that adding more lags made learning harder or introduced overfitting/instability.

Adding more lag steps provides more historical context, but too many lags may reduce generalization if they include noise or irrelevant patterns. The best lag length depends on the problem and should be tuned based on validation performance.
